    <script>
        // Инициализация Telegram WebApp
        let tg = window.Telegram.WebApp;
        tg.expand(); // Разворачиваем окно на весь экран

        // 1. Получаем SubID (Метку канала) из параметров запуска ТГ-бота
        // Например: t.me/ТВОЙ_БОТ?startapp=student_channel
        let subId = tg.initDataUnsafe?.start_param || "organic_traffic";

        // 2. Сюда вставь свои реальные базовые ссылки из Leadgid!
        const offers = {
            'psb': "https://go.leadgid.ru/aff_c?aff_id=149637&offer_id=7249",
            'ozon': "https://go.leadgid.ru/aff_c?aff_id=149637&offer_id=7260&p=adnetwork",
            'vtb': "https://go.leadgid.ru/aff_c?aff_id=149637&offer_id=7332&p=adnetwork"
        };

        // 3. Функция обработки клика по карте
        function openOffer(bankKey) {
            // Берем базовую ссылку
            let baseUrl = offers[bankKey];
            
            // Формируем финальную ссылку с пришитым SubID канала
            // (Проверь в Leadgid, какой параметр они используют, обычно &sub1=)
            let separator = baseUrl.includes("?") ? "&" : "?";
            let finalUrl = baseUrl + separator + "sub1=" + subId;

            // Открываем ссылку во внешнем браузере телефона
            tg.openLink(finalUrl);

            // Если человек выбрал ПСБ или ВТБ - запускаем таймер на 10 секунд (10000 мс) для Апсейла
            if (bankKey === 'psb' || bankKey === 'vtb') {
                setTimeout(() => {
                    document.getElementById('upsellModal').style.display = 'flex';
                }, 10000); 
            }
        }

        // Функция закрытия окна апсейла
        function closeModal() {
            document.getElementById('upsellModal').style.display = 'none';
        }
    </script>
